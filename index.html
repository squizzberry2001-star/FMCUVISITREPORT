Agar aplikasi web dapat menyimpan & membaca Master User, Password, Master Toko, dan Jadwal Visit secara otomatis di project fmcu-visit-database, silakan buka Firebase Console → Firestore Database → Tab Rules, tempel aturan berikut, lalu klik Publish: 

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /{document=**} {
      allow read, write: if true;
    }
  }
}
